cmake_minimum_required(VERSION 3.22)
project(xhynlog LANGUAGES C)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a" OR ANDROID_PLATFORM_LEVEL LESS 30)
  message(FATAL_ERROR "Requires Android NDK, arm64-v8a, API 30 or newer")
endif()
add_library(xhynlog SHARED logcontrol.c)
target_compile_features(xhynlog PRIVATE c_std_11)
target_compile_options(xhynlog PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
target_link_options(xhynlog PRIVATE -Wl,-z,relro,-z,now -Wl,-z,max-page-size=16384)
target_link_libraries(xhynlog PRIVATE dl log)
